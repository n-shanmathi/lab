other commands:

# 1. Check the current user
whoami

# 2. Switch to root user
sudo -i

# 3. Create a normal user
adduser user1

# 4. Set password for the user
passwd user1

# 5. Create a group
groupadd projectgroup

# 6. Add the normal user to the group
usermod -aG projectgroup user1

# 7. Give the user sudo/root privileges
usermod -aG sudo user1

# 8. Check the groups of the user
groups user1

# 9. Switch to the normal user
su - user1

# 10. Check the current user
whoami

# 11. Check whether the user has sudo permission
sudo whoami

# Output should be:
# root


# 12. Create a directory as root
sudo mkdir /project

# 13. Create a file
sudo touch /project/test.txt

# 14. Change the owner of the directory
sudo chown user1:projectgroup /project

# 15. Change the permissions
sudo chmod 770 /project

# 16. Check ownership and permissions
ls -ld /project

# Expected:
# drwxrwx--- user1 projectgroup /project


# 17. Give read, write and execute permission to owner
sudo chmod u+rwx /project

# 18. Give read and write permission to group
sudo chmod g+rw /project

# 19. Remove all permissions from others
sudo chmod o-rwx /project

# 20. Check permissions again
ls -ld /project


# 21. Remove sudo/root privileges from user1 if required
sudo deluser user1 sudo

# 22. Delete the normal user if required
sudo deluser user1

# 23. Delete the group if required
sudo groupdel projectgroup
